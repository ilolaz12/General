#include <xc.h>

#include "lcd.h"
#include "functions.h"

extern enum State currentState;  // Declare the external usage of currentState
extern unsigned int timerFlag;  // Declare the external usage of timerFlag


void main(void) {
    // Call the initialization functions
    init_IO();
    init_adc();
    init_pwm();
    init_LCD();
    init_reset();
    init_buttons();
    init_motor_control();
    init_door();

    unsigned int position = get_position_mm();
    unsigned int nearestFloor = get_pos_floor(pos);
    unsigned int currentFloor = nearestFloor;
    int targetFloor = 0; // Declare and initialize targetFloor.
    int buttonPressed = 0;
    
    while(1) { // endless loop	
        position = get_position_mm();  // Update position inside loop
        nearestFloor = get_pos_floor(position);
        currentFloor = nearestFloor;  // Updated currentFloor with nearestFloor, if applicable
        int b = buttons_in(); 
        display_mm(position);
        display_flr(position);
        __delay_ms(100);
        
        switch (currentState) {
            case IDLE:
                    display_flr(currentFloor); // Refresh the display with current floor
                    int buttonPressed = buttons_out(); // Check if any request button is pressed

                    switch (buttonPressed) {
                        case 1: // Floor 1 button
                        case 2: // Floor 2 button
                        case 3: // Floor 2 button
                        case 4: // Floor 3 button
                            targetFloor = buttonPressed; // Set targetFloor based on the button pressed
                            if (door_check() == DOOR_CLOSED) { // Ensure door is closed before moving
                                if (currentFloor < targetFloor) {
                                    currentState = MOVING_UP;
                                    move(UP_SPEED, UP); // Call move with defined speed and direction
                                } else if (currentFloor > targetFloor) {
                                    currentState = MOVING_DOWN;
                                    move(DOWN_SPEED, DOWN); // Call move with defined speed and direction
                                }
                            }
                            break;
                        case 5: // Door open button
                            if (door_check() == DOOR_CLOSED) { // Only open if door is currently closed
                                door_control(DOOR_OPEN);
                            }
                            break;
                        case 6: // Door close button
                            if (door_check() == DOOR_OPEN) { // Only close if door is currently open
                                door_control(DOOR_CLOSED);
                            }
                            break;
                        default:
                            // Handle no valid button press or maintain IDLE if nothing is pressed
                            break;
                    }
                    break; 
            case MOVING_UP:
                display_flr(currentFloor);
                if (door_check()) {  // Ensure the door is closed before moving
                    move(UP_SPEED, UP);  // Move up with defined speed
                } else {
                    currentState = EMERGENCY;  // Switch to emergency state if door is not closed
                }
                if (currentFloor == targetFloor) {
                    move(0, UP);  // Stop the elevator by setting speed to 0
                    currentState = DOORS_OPEN;
                    door(OPEN);
                    buzzer_control(1);  // Turn on the buzzer
                    delay_ms(250);      // Buzzer on for 250 ms
                    buzzer_control(0);  // Turn off the buzzer
                    start_door_open_timer(8);  // Start door open timer for 8 seconds
                }
                break;
            case MOVING_DOWN:
                display_flr(currentFloor);
                if (door_check()) {
                    move(DOWN_SPEED, DOWN);  // Move down with defined speed
                } else {
                    currentState = EMERGENCY; // Handle an emergency if door opens unexpectedly
                } 
                if (currentFloor == targetFloor) {
                    move(0, UP); // Stop the elevator
                    currentState = DOORS_OPEN;
                    door(OPEN);
                    // Activate the buzzer for 250 ms
                    buzzer_control(1);  // Turn on the buzzer
                    delay_ms(250);      // Delay for 250 ms
                    buzzer_control(0);  // Turn off the buzzer
                    start_door_open_timer(8); // Start the door open timer for 8 seconds
                }
                break;
            case DOORS_OPEN:
                display_flr(currentFloor); // Refresh the display with current floor
                b = buttons_in(); // Check if any request button is pressed
                if (!door_open_timer) { // If the timer is not already running
                    start_door_open_timer(8); // Start 8-second timer for the door being open
                } 
                if (door_o_timer_exp()) { // Check if 8 seconds have passed
                    door(CLOSE); // Close the door
                    if (currentFloor < targetFloor) {
                        currentState = MOVING_UP;
                    } else if (currentFloor > targetFloor) {
                        currentState = MOVING_DOWN;
                    } else {
                    currentState = IDLE; // Transition to the IDLE state
                    }
                    door_open_timer = 0; // Reset the door-open timer
                }
                break;
            case EMERGENCY: // Handle emergency situations
                motor_stop(); // Immediately stop any movement
                // Check the current position and determine the nearest floor
                if (currentFloor != nearestFloor) {
                    move(nearestFloor); // If not already at a floor, move to the nearest floor
                }
                door(OPEN); // Open the doors once the floor is reached or if already at a floor
                // Display emergency message on LCD
                lcd_clear();
                //lcd_set_cursor(0, 0);
                lcd_str("EMERGENCY");
                //lcd_set_cursor(1, 0);
                lcd_str("Do not use elevator");
                // Reset to IDLE or a safe state after emergency is cleared
                currentState = IDLE;
                break;
            case MAINTENANCE:
                // Check for maintenance mode trigger
                if (currentFloor == 1 && open_button_pin == 1 && close_button_pin == 1) {
                    currentState = MAINTENANCE;
                }
                lcd_clear();
                lcd_str("Maintenance Mode");
                // Move to floor 3
                targetFloor = 3;
                move(targetFloor, UP); 
                while (currentFloor != targetFloor) {   
                    display_mm(pos); // Continuously display the current position
                    __delay_ms(100); // Short delay for readability and CPU relief
                }
                // Move back to floor 1
                targetFloor = 1;
                move(targetFloor, UP);
                while (currentFloor != targetFloor) {
                    display_mm(pos); // Continuously display the current position
                    __delay_ms(100); // Short delay for readability and CPU relief
                }
                // Return to normal operation
                lcd_clear();
                lcd_str("Maintenance Complete");
                currentState = IDLE;
                break;
            }
        if (timerFlag) { // Timer-based actions
            timerFlag = 0; // Reset the timer flag
        }
    }
}

void setup_timer(void) {
    T0CON = 0x07; // 16-bit, internal clock, prescaler 1:256
    TMR0 = 65536 - (unsigned int)((5000 * (_XTAL_FREQ / 4 / 256)) % 65536); // Calculate and set TMR0
    INTCONbits.TMR0IE = 1;  // Enable Timer0 interrupt
    INTCONbits.TMR0IF = 0;  // Clear Timer0 interrupt flag
    T0CONbits.TMR0ON = 1;  // Turn on Timer0
}

void __interrupt(high_priority) high_priority_isr(void) {

        timer_expired_flag = 1; // Set timer expiration flag

        INTCONbits.TMR0IF = 0; // Clear Timer0 interrupt flag

        currentState = EMERGENCY; // Assume you have proper handling here
        INTCON3bits.INT1IF = 0; // Clear interrupt flag
    }
}

void start_door_open_timer(int seconds) { // Start the door-open timer

void move(unsigned int speed, enum Direction dir) {
    if (dir == STOPPED) { // Explicitly stop the motor
        motor_stop();
    } else if (speed <= 100) {
        // Calculate the duty cycle for PWM based on speed
        unsigned char dutyCycle = (unsigned char)((speed * 255u) / 100u);
        // Set PWM duty cycle
        CCPR1L = dutyCycle;
        // Set the direction pin
        switch (dir) {
            case UP:
                UPDN = 1;  // UPDN is a pin or register setting for motor direction upwards
                break;
            case DOWN:
                UPDN = 0;  // UPDN is a pin or register setting for motor direction downwards
                break;
            default:
                motor_stop();  // Safely handle any unexpected direction values
                break;
        }
    } else {
        // If the speed is out of bounds, stop the motor
        motor_stop();
    }
}

void init_IO() {
    TRISB = 0xFF; // Set PORTB (RB0-RB7) as inputs for buttons and door ajar
    TRISD = 0x00; // Set PORTD (RD0-RD7) as outputs for display
    TRISC = 0b11111000; // RC0, RC1, and RC2 as outputs for buzzer, PWM, and motor control
    TRISD = 0x00; // PORTD as output for LCD Data
    TRISA = 0xFF; //Initialize PORTA for inputs: Up/Down Control and Open/Close Push Buttons

    // Position sensor - analog input
    ADCON0 = 0x01;  // AN0 selected, turn on ADC
    ADCON1 = 0x0E;  // Configure AN0 as analog and the rest as digital
    ADCON2 = 0xA9; // Acquisition time and conversion clock
    
    // Door Control Relay
    TRISEbits.TRISE0 = 0; // Set RE0 as output for Door Control Relay
    INTCON2bits.RBPU = 0; // Enable pull-ups on PORTB for switches
    
    // Buzzer setup
    LATCbits.LATC2 = 0;  // Initial state to low/off
    TRISCbits.TRISC2 = 0; // Set the RC2 pin as an output

    // Winch Unit + init_pwm
    // up/down - Digital output
    TRISD0 = 0; // Set RD0 as output for Up
    TRISD1 = 0; // Set RD1 as output for Down
    TRISC2 = 0;     // Set CCP1 pin as output
    // Initialize PWM for speed control
    PR2 = _PR2_VALUE_; // Set PWM frequency
    CCP1CON = 0x0C; // PWM mode
    T2CON = 0x04; // TMR2 ON
    CCPR1L = 0x1E; // Initial duty cycle
    // RCSTAbits.SPEN = 1; //enable serial port
    // TXSTAbits.SYNC = 0; //clean sync bit
    // TRISC = 0b10000000; //setTX and RX pins
    
    // set baud rate 9600 baud with FOSC 1MHZ
    BAUDCONbits.BRG16 = 1;
    TXSTAbits.BRGH =1;
    SPBRG = 25;
     // LCD setup
    TRISEbits.TRISE1 = 0; // RE1 as output for LCD EN
    TRISEbits.TRISE2 = 0; // RE2 as output for LCD RS
    LATEbits.LATE1 = 0;  // Ensure LCD EN is low
    LATEbits.LATE2 = 0;  // Ensure LCD RS is low   
    
    TXSTAbits.TXEN = 1; //enable the transmitter
    TRISEbits.TRISE0 = 1; // Set RE0 as input for Circuit Breaker
    TRISCbits.TRISC7 = 1; // Set RC7 as input for Emergency Signal
    
    setup_timer(); // Configure the hardware timer
}

void init_adc(void){ // digital inputs
    // Configure the ADC
    ADCON1 = 0x0E;  // Configure AN0 as analog and the rest as digital
    ADCON0 = 0x01;  // AN0 selected, turn on ADC
    ADCON2 = 0xA9;  // Acquisition time and conversion clock settings
    // Set the TRIS register for the analog pin
    TRISAbits.TRISA0 = 1; // Set RA0/AN0 as input for the Position Sensor
    while(adc_ready == 0) {
        __delay_us(100);  // Wait for ADC to be ready
    }
}

void init_pwm(void) {
    // PWM on RC2 (CCP1)
    TRISCbits.TRISC2 = 0; // Set RC2 as output for PWM
    PR2 = 254;            // Set PWM period
    T2CON = 0x07;         // Timer2 on, prescaler settings as needed
    CCP1CON = 0x0C;       // PWM mode on CCP1
    CCPR1L = 0;           // Set PWM duty cycle: initialise speed to stop
    // Configure motor direction control pins as output
    TRISCbits.TRISC0 = 0; // Set RC0 as an output
    TRISCbits.TRISC1 = 0; // Set RC1 as an output
    // Configure motor direction control pins as output
    TRISCbits.TRISC0 = 0; // Set RC0 as an output
    TRISCbits.TRISC1 = 0; // Set RC1 as an output
}
unsigned int read_adc(unsigned int ch) {
    while(adc_ready == 0){ // Wait for previous ADC read to complete
        ADCON0bits.CHS = (unsigned char)ch;   // Set the ADC channel
        ADCON0bits.GO = 1;     // Start the conversion
        adc_ready = 0;         // Clear the ready flag
    }
    return adc_result;     // Return the ADC result
}

void lcd_cmd(unsigned char cmd) {
    LCD_DATA = cmd;   // Send command to data port
    LCD_RS = 0;       // Set RS to 0 for command mode
    LCD_EN = 1;       // Enable pulse start
    __delay_ms(1);    // Delay for processing
    LCD_EN = 0;       // End pulse
}

void lcd_clear(void) {
    lcd_cmd(0x01);  // Command to clear display
    __delay_ms(2);  // Delay to complete clear
}

// Function to display a string on the LCD
void lcd_str(const char* s) {
    while(*s) {  // Loop through each character of the string
        lcd_data(*s++);  // Increment pointer to next character
    }
}

void init_LCD() {
    TRISD = 0x00; // Set up data port as output
    TRISEbits.RE1 = 0;  // Set RE1 as output for RS
    TRISEbits.RE2 = 0;  // Set RE2 as output for Enable

    // Basic initialization sequence for HD44780 LCD
    __delay_ms(20);    // Wait for more than 15ms after Vcc rises to 4.5V
    lcd_cmd(0x30);     // Function set: 8-bit, 1 line, 5x8 dots
    __delay_ms(5);     // Wait more than 4.1ms
    lcd_cmd(0x30);     // Function set: 8-bit, 1 line, 5x8 dots
    __delay_us(100);   // Wait more than 100µs
    lcd_cmd(0x30);     // Function set: 8-bit, 1 line, 5x8 dots

    lcd_cmd(0x38);     // Function set: 8-bit, 2 lines, 5x8 dots
    lcd_cmd(0x10);     // Set cursor
    lcd_cmd(0x0C);     // Display ON; Cursor OFF
    lcd_cmd(0x06);     // Entry mode set

    __delay_ms(20);
    lcd_cmd(LCD_CFG);
    lcd_cmd(LCD_ON);
    lcd_clear();
    lcd_cmd(0x06);  // Entry Mode Set
}
// Implement lcd_puts for string output
void lcd_puts(const char* s) {
    while(*s) {
        lcd_data(*s++);
    }
}
// Function to send data to the LCD
void lcd_data(unsigned char data) {
    LCD_RS = 1;         // Set RS to 1 for data mode
    LCD_DATA = data;    // Send data to data port
    LCD_EN = 1;         // Enable pulse
    __delay_us(40);     // Data write delay
    LCD_EN = 0;         // End pulse
}

void motor_direction(int direction) {
    switch (direction) {
        case UP:
            UPDN = 1;  // Set the direction pin for upward movement
            SPEED = UP_SPEED;  // Set speed for upward movement
            break;
        case DOWN:
            UPDN = 0;  // Set the direction pin for downward movement
            SPEED = DOWN_SPEED;  // Set speed for downward movement
            break;
        case STOP:
            SPEED = 0;  // Set speed to 0 to stop the motor
            // Optional: Explicitly set UPDN to a safe state if required
            break;
        default:
            UPDN = 0;  // Ensure motor is stopped if direction is undefined
            SPEED = 0;  // Ensure speed is set to 0
            break;
    }
}
void motor_stop() { // Stops the motor
    motor_direction(STOP);
}
// Adjust the speed control for motor.
void motor_speed_control(int speed) {
    if (speed < 0 || speed > 100) return;  // Guard clause for speed range

    unsigned char dutyCycle = (unsigned char)((speed * 255u) / 100u);  // Convert to 8-bit value properly
    CCPR1L = dutyCycle >> 2;  // Assign high byte
    CCP1CON = (CCP1CON & 0xCF) | ((dutyCycle & 0x03) << 4);  // Handle low byte
}

void move_to_floor(int floor) {
    if (currentFloor == floor) return; // Early exit if already on the correct floor
    door_control(DOOR_CLOSED);  // Ensure door is closed before moving
    enum Direction direction = (currentFloor < floor) ? UP : DOWN; // Use enum type for direction
    move(100, direction); // Move with maximum speed
    while (currentFloor != floor) {
        update_position(); // Update current floor based on sensor input
    }
    move(0, STOP);  // Stop the elevator on arrival
    door_control(DOOR_OPEN);  // Open door after stopping
}

unsigned int get_pos_floor(unsigned int pos) {
    int which = 0;
    if (pos < (FLR1 - TOL)) {
        which = -1; // too low
    } else if (pos <= (FLR1 + TOL)) {
        which = 1;
    } else if (pos <= (FLR2 + TOL)) {
        which = 2;
    } else if (pos <= (FLR3 + TOL)) {
        which = 3;
    } else {
        which = -1; // too high or between floors
    }
    return which; // Return the floor number
}

    unsigned int position_mm = get_position_mm(); // Get current position in millimeters.
    int currentFloor = get_pos_floor(position_mm); // Update the current floor based on new sensor data.
}
